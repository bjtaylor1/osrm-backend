# https://serverfault.com/questions/958965/nginx-enabling-cors-for-multiple-subdomains

map $http_origin $allow_origin {
	~^https://gpxplanner.app$ $http_origin;
	~^https://(www\.)?gpxeditor.co.uk$ $http_origin;
	~^https?://localhost(:\d+)?$ $http_origin;
	~^https?://192\.168\.0\.(\d+)(:\d+)?$ $http_origin;
	default "";
}

	location ~ ^/match {
		if ($request_method = OPTIONS) {
			add_header Access-Control-Allow-Origin $allow_origin always;
			add_header Access-Control-Allow-Methods "GET, OPTIONS" always;
			add_header Access-Control-Allow-Headers "Authorization, Content-Type, X-Requested-With" always;
			add_header Access-Control-Max-Age 86400 always;
			add_header Vary Origin always;
			return 204;
		}

		# Strip token param before proxying to OSRM (which rejects unknown params)
		access_by_lua_block {
			local args = ngx.var.args or ""
			args = args:gsub("[&?]?token=[^&]*", "")
			args = args:gsub("^&", "")
			ngx.var.args = args
		}

		rewrite ^/match[^/]*/(.*) /match/$1 break;
		proxy_pass http://127.0.0.1:5001;
		proxy_set_header Host $host;
		proxy_set_header X-Real-IP $remote_addr;

		add_header Vary Origin always;
	}

			add_header Access-Control-Max-Age 86400 always;

		# Strip token param before proxying to OSRM (which rejects unknown params)
		access_by_lua_block {
